# Small code/data changes to 0EXEC.BIN, applied by tools\build.ps1 (via codepatch.exe) after the text.
# Format: file-offset  original-bytes  new-bytes  comment     (file offset = RAM address - 0x06010000)
# codepatch refuses any change whose original bytes don't match, so a wrong offset can't slip through.
# To switch a change off, put '#' in front of its line(s). Changes in one group belong together.
#
# Free space used for new code/data: 0x315B0-0x315F7, which held the text of a debug-only
# animation viewer ("ごっちアニメみてみてチェック" etc.; drawn only by debug code at 0x060229F0-0x06022A3C).

# --- A. Names end in the squeezed "TCHI" ([ = TC, ] = HI; see font-fixes.txt) instead of "っち" everywhere.
# The name builder at 0x0602AA84 appends っ (AF) and ち (C1). We jump to a new routine that writes [ and ]
# (the 30-byte name buffer is cleared first, so the result stays terminated), then jump back.
0x1AB7E 6073931477019213376C06342720 D30B432B60730009000900090009 jump to the new routine (index in r0)
0x1ABAC 00AF00C1 060415B0 the jump target address, stored where the っ/ち values were
0x315B0 BADEAFC140B1C6D240D0C3D0C340C1AAAFB840004C52C3DEBADEAFC140CADEB0 E35B06347001E35D0634D301432B00090602AB8C000900090009000900090009 new routine: write [ ], jump back to 0x0602AB8C

# --- B. The name entry screen starts in ABC (English) mode instead of hiragana
# Mode byte at 0x06082AA4: 0 = hiragana, 1 = katakana, 2 = English. The setup routine (0x0602ABB4) now
# stores 2 there (and still sets 0x06082AA5 to 0xFF), and the caller's second "mode = 0" is removed.
0x1ABBC 23E0923AD11F2120 E1022310E0FF8031 setup: mode = 2; byte after it = 0xFF
0x19B3E 22C0 0009 caller no longer resets mode to 0

# --- C. (switched off) Name entry: name, cursor and ending shifted 10 pixels left, needed when TCHI took
# 4 letter spaces. With the squeezed 2-space TCHI the original layout fits, so these stay off.
#0x1AA34 E57C E572 name x 124 -> 114
#0x1A252 707C 7072 cursor x 124 -> 114
#0x1AA62 00CA 00C0 name ending x 202 -> 192

# --- D. Top bar day counter: digits move from positions 2-4 to 4-6 of the line, so it can read "DAY   1"

# --- E. Save/load screen day counter: digits move from positions 0-2 to 3-5 (template "DAY   " in

0x2034A 7244 7247 hundreds digit -> stack+71
0x2035C 7345 7348 tens digit -> stack+72
0x20362 7146 7149 ones digit -> stack+73
